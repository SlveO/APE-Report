"""Authorized, resumable public HTTPS acquisition; never computes file digests."""
import argparse

from datetime import datetime, timezone
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
LOG = ROOT / 'logs/runs/20261113-main80-assets'
RAW = ROOT / 'datasets/raw-main80'
MODEL = ROOT / 'models/Llama-3.1-8B-Instruct'
LOCK = threading.Lock()


def now():
    return datetime.now(timezone.utc).isoformat()


def event(record):
    with LOCK, (LOG / 'events.jsonl').open('a') as f:
        f.write(json.dumps({'time': now(), **record}, ensure_ascii=False) + '\n')
        f.flush()
        os.fsync(f.fileno())


def get_json(url, target):
    fetch(url, target, 'source metadata; remote mutable branch', expected=None)
    return json.loads(target.read_text())


def fetch(url, target, license_note, expected=None):
    target = Path(target)
    assert target.is_relative_to(ROOT) and target.resolve().is_relative_to(ROOT)
    target.parent.mkdir(parents=True, exist_ok=True)
    if target.exists():
        if expected is not None and target.stat().st_size != expected:
            raise RuntimeError('existing target size differs; preserve and inspect: ' + str(target.relative_to(ROOT)))
        event({'action': 'reuse_existing', 'path': str(target.relative_to(ROOT)), 'bytes': target.stat().st_size})
        return target
    event({'action': 'source_declared_before_fetch', 'url': url, 'target': str(target.relative_to(ROOT)), 'format': target.suffix, 'license': license_note, 'failure_policy': 'retain_partial_no_substitution', 'expected_bytes': expected})
    partial = target.with_name(target.name + '.partial')
    for attempt in range(1, 4):
        offset = partial.stat().st_size if partial.exists() else 0
        if expected is not None and offset == expected:
            partial.rename(target)
            return target
        headers = {'User-Agent': 'APE-MAIN80-research-acquisition/1.0'}
        if offset:
            headers['Range'] = f'bytes={offset}-'
        try:
            with urllib.request.urlopen(urllib.request.Request(url, headers=headers), timeout=45) as r:
                if offset and r.status != 206:
                    abandoned = partial.with_name(partial.name + f'.range-ignored-{time.time_ns()}')
                    partial.rename(abandoned)
                    event({'action': 'range_ignored_preserved', 'path': str(abandoned.relative_to(ROOT)), 'bytes': offset})
                    offset = 0
                response_length = r.headers.get('Content-Length')
                event({'action': 'response', 'target': str(target.relative_to(ROOT)), 'attempt': attempt, 'http_status': r.status, 'offset': offset, 'content_length': response_length, 'content_range': r.headers.get('Content-Range'), 'content_type': r.headers.get('Content-Type')})
                written = offset
                last_report = time.monotonic()
                with partial.open('ab' if partial.exists() else 'xb') as f:
                    while True:
                        chunk = r.read(8 * 1024 * 1024)
                        if not chunk:
                            break
                        f.write(chunk)
                        written += len(chunk)
                        if time.monotonic() - last_report >= 30:
                            f.flush()
                            event({'action': 'progress', 'target': str(target.relative_to(ROOT)), 'bytes': written})
                            last_report = time.monotonic()
                    f.flush()
                    os.fsync(f.fileno())
                if response_length is not None and written - offset != int(response_length):
                    raise RuntimeError('response body length mismatch')
                if expected is not None and written != expected:
                    raise RuntimeError('expected file size mismatch')
                partial.rename(target)
                event({'action': 'completed', 'target': str(target.relative_to(ROOT)), 'bytes': written, 'attempt': attempt})
                print(json.dumps({'completed': str(target.relative_to(ROOT)), 'bytes': written}), flush=True)
                return target
        except Exception as e:
            event({'action': 'failure', 'target': str(target.relative_to(ROOT)), 'attempt': attempt, 'exception': type(e).__name__, 'http_code': getattr(e, 'code', None), 'reason_type': type(getattr(e, 'reason', None)).__name__})
            if attempt == 3:
                raise RuntimeError(f'acquisition failed: {target.relative_to(ROOT)} ({type(e).__name__}, HTTP {getattr(e, "code", None)})') from None
            time.sleep(min(2 ** attempt, 8))


def model(include_weights=True):
    url = 'https://modelscope.cn/api/v1/models/LLM-Research/Meta-Llama-3.1-8B-Instruct/repo/files?Revision=master&Recursive=true'
    metadata = get_json(url, LOG / 'model-files-source.json')
    files = {r['Path']: r for r in metadata['Data']['Files'] if r['Type'] == 'blob'}
    required = ['LICENSE', 'README.md', 'USE_POLICY.md', 'config.json', 'generation_config.json', 'model.safetensors.index.json', 'special_tokens_map.json', 'tokenizer.json', 'tokenizer_config.json'] + [f'model-{i:05d}-of-00004.safetensors' for i in range(1, 5)]
    assert len(required) == 13 and all(name in files for name in required)
    note = 'Meta Llama 3.1 Community License + USE_POLICY; ModelScope user-uploaded LLM-Research mirror, not official Meta release; master is mutable retrieval branch'
    manifest = {'source': 'LLM-Research/Meta-Llama-3.1-8B-Instruct', 'fetch_branch': 'master', 'immutable_revision_claimed': False, 'official_meta_publication_claimed': False, 'format': 'safetensors', 'license': note, 'observed_at': now(), 'required_files': [{k: files[n].get(k) for k in ('Path', 'Size', 'Revision')} for n in required]}
    (LOG / 'model-manifest.json').write_text(json.dumps(manifest, indent=2))
    def one(name):
        endpoint = 'https://modelscope.cn/api/v1/models/LLM-Research/Meta-Llama-3.1-8B-Instruct/repo?' + urllib.parse.urlencode({'Revision': 'master', 'FilePath': name})
        return fetch(endpoint, MODEL / name, note, files[name]['Size'])
    for name in required[:9]:
        one(name)
    if not include_weights:
        return
    with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:
        for result in pool.map(one, required[9:]):
            pass
