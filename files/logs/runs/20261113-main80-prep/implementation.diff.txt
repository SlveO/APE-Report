+def bucket(length):
+    return min((4096, 8192, 16384, 32768, 65536, 131072), key=lambda x: abs(math.log2(max(length or 1, 1) / x)))

+def speed_key(method, length):
+    return ('ape' if method.startswith('ape_') else method, bucket(length))

+def rates(records):
+    values = defaultdict(list)
+    for r in records:
+        if r['status'] == 'success' and r['key']['stage'] != 'attention':
+            # Cold and hot requests cannot share a latency estimate.
+            cache = 'hot_recovered' if r.get('timing_condition') == 'recovered_application_GPU_KV' else r.get('cache_mode', 'cold')
+            key = (*speed_key(r['method'], r['L']), cache)
+            values[key].append(r['total_seconds'])
+    return [{'method': k[0], 'length_bucket': k[1], 'cache_mode': k[2], 'count': len(v),
+             'mean': mean(v), 'low': min(v), 'high': max(v)} for k, v in sorted(values.items())]

+def estimate(jobs, ledger, speed, conservative=False, partial=False):
+    total_low, total_high = 0.0, 0.0
+    observed_jobs = 0
+    by_key = {(r['method'], r['length_bucket'], r['cache_mode']): r for r in speed if r['count'] >= 3}
+    for job in jobs:
+        if ledger.finished(job['key']):
+            continue
+        method, length = speed_key(job['method'], job['key']['L'])
+        cache = job['cache']
+        observation = by_key.get((method, length, cache))
+        proxy = False
+        if observation is None and conservative and cache in {'hot', 'search'}:
+            observation = by_key.get((method, length, 'cold'))
+            proxy = observation is not None
+        if observation is None:
+            # No invented speed for an unobserved method/length/cache stratum.
+            if partial:
+                continue
+            return None
+        observed_jobs += 1
+        total_low += 0 if proxy else observation['low']
+        total_high += observation['high']
+    return (total_low, total_high) if observed_jobs or not partial else None
+

+def all_jobs(factory, limits, candidates):
+    # Conservative validation count before exploration completes (three distinct choices).
+    temporary = candidates or {'global': [0.5, 0.5], 'cells': {f'{l}/{n}': [0.7, 0.7] for l in (4096, 8192, 16384) for n in (4, 8, 16)}}
+    return [j for stage in STAGES for j in factory.make(stage, limits, temporary)]

+def selected_records(ledger, jobs, directory):

+                'sample_cell': group, 'missing_keys': [k for k in keys if ledger.outcome(k) is None],
+                'basis': 'failed or missing output; never score-based, no replacement'})
+    return paired

+def reduce_budget(directory, state, factory, candidates):

+            'reason': 'observed method/length stratified speed; no score used'})

+def inside(path):

+def write_json(path, value):
+    path = inside(path)
+    path.parent.mkdir(parents=True, exist_ok=True)
+    tmp = path.with_name(path.name + '.tmp')
+    with tmp.open('w') as f:
+        json.dump(value, f, ensure_ascii=False, indent=2, allow_nan=False)
+        f.write('\n')
+        f.flush()
+        os.fsync(f.fileno())
+    tmp.replace(path)

+def identity(value):
+    """A reversible representation, never a digest."""
+    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(',', ':'), allow_nan=False)

+def append_json(path, value):
+    path = inside(path)
+    path.parent.mkdir(parents=True, exist_ok=True)
+    # Preserve even a torn last record. The reader reports it, never truncates it.
+    with path.open('ab+') as f:
+        if f.tell():
+            f.seek(-1, os.SEEK_END)
+            if f.read(1) != b'\n':
+                f.seek(0, os.SEEK_END)
+                f.write(b'\n')
+        f.seek(0, os.SEEK_END)
+        f.write((identity(value) + '\n').encode())
+        f.flush()
+        os.fsync(f.fileno())

+"""Diagnostic-only token probabilities, checked against actual APE output.
+
+The branch masses alone are not token attention. Reconstruct the within-branch
+softmax, multiply by actual BF16 branch mass, then compare weighted values.
+"""
+import math
+
+import torch
+
+from .common import append_json
+
+
+class AttentionCapture:
+    def __init__(self, model, output, metadata):
+        self.model, self.output, self.metadata = model, output, metadata
+        self.step = {}
+
+    def __enter__(self):
+        for i in (0, 15, 31):
+            attention = self.model.model.layers[i].self_attn
+            attention.ape_observer = self.callback(i)
+        return self
+
+    def callback(self, layer):
+        def capture(module, q, kc, ko, vc, vo, mass, actual, temperature):
+            step = self.step.get(layer, 0)
+            self.step[layer] = step + 1
+            heads = [0, 8, 16, 24]
+            qi = q[0, heads, -1].float()
+            context_logits = torch.einsum('hd,hkd->hk', qi, kc[0, heads].float()) / (math.sqrt(module.head_dim) * temperature)
+            other_logits = torch.einsum('hd,hkd->hk', qi, ko[0, heads].float()) / math.sqrt(module.head_dim)
+            # The last query position sees every cached/request token (causal tail).
+            pc, po = context_logits.softmax(-1), other_logits.softmax(-1)
+            branch = mass[0, -1, heads, 0].float()
+            wc, wo = pc * branch[:, :1], po * branch[:, 1:]
+            reconstruction = torch.einsum('hk,hkd->hd', wc, vc[0, heads].float()) + torch.einsum('hk,hkd->hd', wo, vo[0, heads].float())
+            target = actual[0, -1, heads].float()
+            max_abs = float((reconstruction - target).abs().max())
+            # Quantized flash outputs and BF16 branch mixing introduce rounding.
+            tolerance = 0.03 + 0.02 * float(target.abs().max())
+            if max_abs > tolerance:
+                raise RuntimeError(f'token attention reconstruction mismatch: {max_abs} > {tolerance}')
+            p = module.len_prefix
+            # Restore physical cache order: shared prefix, all documents, request.
+            weights = torch.cat([wo[:, :p], wc, wo[:, p:]], dim=-1)
+            append_json(self.output, {**self.metadata, 'layer': layer, 'heads': heads, 'step': step,
+                'query_kind': 'query_last_token' if step == 0 else 'decode_token',
+                'max_abs_output_error': max_abs, 'tolerance': tolerance,
+                'token_probabilities': weights.cpu().tolist(), 'branch_masses': branch.cpu().tolist()})
+        return capture

+"""One model per worker. PCW uses ordinary causal Llama, never APE patches.
+
+PCW merge/positions follow AI21Labs/Parallel-Context-Windows/pcw_wrapper.py.
+The stock Transformers 4.44.2 Llama implements GQA and Llama-3.1 RoPE.
+"""
+import random
+import time
+from dataclasses import dataclass
+
+import torch
+from transformers import AutoModelForCausalLM, AutoTokenizer
+
+from ape import enable_attention_prefill_prefix, enable_attention_prefill_context, enable_attention_prefill_query
+from .rules import DEFAULT, choose
+
+METHODS = ('sequential', 'pcw', 'ape_default', 'ape_fixed', 'ape_dynamic')
+
+
+class SampleError(ValueError):
+    def __init__(self, message, evidence=None):
+        super().__init__(message)
+        self.evidence = evidence or {}
+
+
+def chunks(tokens, count=None, maximum=7500):
+    if not tokens:
+        raise SampleError('empty document')
+    if count is None:
+        return [tokens[i:i + maximum] for i in range(0, len(tokens), maximum)]
+    if not 1 <= count <= len(tokens):
+        raise SampleError('invalid actual chunk count')
+    q, r = divmod(len(tokens), count)
+    lengths = [q + (i < r) for i in range(count)]
+    result, start = [], 0
+    for length in lengths:
+        result.append(tokens[start:start + length])
+        start += length
+    assert sum(result, []) == tokens
+    return result
+
+
+def merge_pcw(windows, lengths):
+    """Longest first (stable tie), one BOS; KV remains in compact GQA layout."""
+    longest = max(range(len(lengths)), key=lengths.__getitem__)
+    order = [longest] + [i for i in range(len(lengths)) if i != longest]
+    cache = tuple(tuple(torch.cat([windows[i][layer][kv][:, :, (0 if i == longest else 1):, :] for i in order], dim=2)
+                        for kv in (0, 1)) for layer in range(len(windows[0])))
+    return cache, max(lengths), order
+
+
+def merge_ape(prefix, windows):
+    p = prefix[0][0].shape[2]
+    return tuple((torch.cat([pre[0]] + [w[layer][0][:, :, p:] for w in windows], dim=2),
+                  torch.cat([pre[1]] + [w[layer][1][:, :, p:] for w in windows], dim=2),
+                  torch.cat([pre[2]] + [w[layer][2][:, p:] for w in windows], dim=1),
+                  len(windows)) for layer, pre in enumerate(prefix))
+
+
+@dataclass
+class Encoded:
+    cache: tuple
+    next_position: int
+    request_tokens: list
+    order: list
+    document_length: int
+    block_lengths: list
+    cache_key: tuple
+
+
+class Engine:
+    def __init__(self, model, tokenizer, method, *, device='cuda:0'):
+        if method not in METHODS:
+            raise ValueError(method)
+        self.model, self.tokenizer, self.method = model.eval(), tokenizer, method
+        self.device = torch.device(device)
+        self.encoded = None
+        self.stage_calls = {'prefix': 0, 'context': 0, 'query': 0}
+        self.stage = None
+        self.hook = model.model.layers[0].self_attn.register_forward_hook(self._observed)
+
+    def _observed(self, module, args, output):
+        if self.stage is not None:
+            self.stage_calls[self.stage] += 1
+
+    @classmethod
+    def load(cls, path, method):
+        random.seed(42)
+        torch.manual_seed(42)
+        torch.cuda.manual_seed(42)
+        tokenizer = AutoTokenizer.from_pretrained(str(path), local_files_only=True, trust_remote_code=False)
+        model = AutoModelForCausalLM.from_pretrained(str(path), local_files_only=True, trust_remote_code=False,
+                    use_safetensors=True, torch_dtype=torch.bfloat16, low_cpu_mem_usage=True,
+                    attn_implementation='flash_attention_2').to('cuda:0').eval()
+        return cls(model, tokenizer, method)
+
+    def sync(self):
+        if self.device.type == 'cuda':
+            torch.cuda.synchronize(self.device)
+
+    def tensor(self, tokens):
+        return torch.tensor([tokens], device=self.device, dtype=torch.long)
+
+    def forward(self, tokens, cache=None, position=None, logits=False):
+        inputs = self.tensor(tokens)
+        kwargs = {'input_ids': inputs, 'past_key_values': cache, 'use_cache': True, 'return_dict': True}
+        if position is not None:
+            kwargs['position_ids'] = torch.arange(position, position + len(tokens), device=self.device).unsqueeze(0)
+        output = self.model.model(**kwargs)
+        last = self.model.lm_head(output.last_hidden_state[:, -1:, :])[:, -1, :] if logits else None
+        return output.past_key_values, last
+
+    def prepare(self, row):
+        """Pre-tokenized dev documents remain exact; no chat BOS is inferred."""
+        encode = lambda text: self.tokenizer.encode(text, add_special_tokens=False)
+        prefix = encode(row['prefix'])
+        suffix = encode(row['suffix'])
+        document = row['document_tokens'] if 'document_tokens' in row else encode(row['document'])
+        if not prefix or prefix[0] != self.tokenizer.bos_token_id or prefix.count(self.tokenizer.bos_token_id) != 1:
+            raise SampleError('common prefix must contain exactly one initial BOS')
+        if self.tokenizer.bos_token_id in document + suffix:
+            raise SampleError('unexpected BOS inside document/request')
+        parts = chunks(document, count=row.get('N'))
+        if row.get('L') is not None and len(document) != row['L']:
+            raise SampleError('frozen exact length mismatch')
+        if len(prefix) + len(document) + len(suffix) + row['max_new_tokens'] > self.model.config.max_position_embeddings:
+            raise SampleError('full public input exceeds model window; no silent truncation')
+        return prefix, parts, suffix
+
+    def encode(self, prefix, parts, suffix, cache_key):
+        self.stage = None
+        if self.method == 'sequential':
+            cache, _ = self.forward(prefix + sum(parts, []))
+            return Encoded(cache, len(prefix) + sum(map(len, parts)), suffix, list(range(len(parts))), sum(map(len, parts)), list(map(len, parts)), cache_key)
+        if self.method == 'pcw':
+            # Each independent window starts with its own BOS. The request has none.
+            windows = [self.forward([self.tokenizer.bos_token_id] + p, position=0)[0] for p in parts]
+            cache, position, order = merge_pcw(windows, [len(p) + 1 for p in parts])
+            return Encoded(cache, position, prefix[1:] + suffix, order, sum(map(len, parts)), list(map(len, parts)), cache_key)
+        enable_attention_prefill_prefix('llama3.1-8b-instruct', self.model)
+        self.stage = 'prefix'
+        pre, _ = self.forward(prefix)
+        enable_attention_prefill_context('llama3.1-8b-instruct', self.model)
+        self.stage = 'context'
+        windows = [self.forward(p, cache=pre, position=len(prefix))[0] for p in parts]
+        cache = merge_ape(pre, windows)
+        return Encoded(cache, len(prefix) + max(map(len, parts)), suffix, list(range(len(parts))), sum(map(len, parts)), list(map(len, parts)), cache_key)
+
+    @torch.inference_mode()
+    def restore_cache(self, row, param=DEFAULT, rule=None):
+        """Rebuild only lost document KV on resume, without another answer."""
+        self.sync()
+        started = time.perf_counter()
+        prefix, parts, suffix = self.prepare(row)
+        if self.method == 'ape_dynamic':
+            param = choose(rule, sum(map(len, parts)), len(parts))
+        elif self.method == 'ape_default':
+            param = DEFAULT
+        elif self.method == 'ape_fixed' and rule is not None:
+            param = tuple(rule['fixed'])
+        key = (self.method, tuple(prefix if self.method != 'pcw' else [self.tokenizer.bos_token_id]),
+               tuple(tuple(p) for p in parts), tuple(param))
+        self.encoded = self.encode(prefix, parts, suffix, key)
+        self.sync()
+        return time.perf_counter() - started
+
+    @torch.inference_mode()
+    def run(self, row, *, param=DEFAULT, rule=None, cache_mode='cold'):
+        if cache_mode not in {'cold', 'hot', 'search'}:
+            raise ValueError(cache_mode)
+        self.sync()
+        if self.device.type == 'cuda':
+            torch.cuda.reset_peak_memory_stats(self.device)
+        started = time.perf_counter()
+        before_calls = self.stage_calls.copy()
+        prefix, parts, suffix = self.prepare(row)
+        select_start = time.perf_counter()
+        if self.method == 'ape_dynamic':
+            if rule is None:
+                raise ValueError('dynamic rule is not frozen')
+            param = choose(rule, sum(map(len, parts)), len(parts))
+        elif self.method == 'ape_default':
+            param = DEFAULT
+        elif self.method == 'ape_fixed' and rule is not None:
+            param = tuple(rule['fixed'])
+        selection_seconds = time.perf_counter() - select_start
+        # KV encoding for APE is T/S-independent. Search reuse is explicitly labelled;
+        # application cache identities include actual configuration and method.
+        encoding_param = None if cache_mode == 'search' else tuple(param)
+        key = (self.method, tuple(prefix if self.method != 'pcw' else [self.tokenizer.bos_token_id]),
+               tuple(tuple(p) for p in parts), encoding_param)
+        get_start = time.perf_counter()
+        hit = cache_mode != 'cold' and self.encoded is not None and self.encoded.cache_key == key
+        cache_get_seconds = time.perf_counter() - get_start
+        if cache_mode == 'hot' and not hit:
+            raise ValueError('hot request has no matching in-process GPU cache')
+        build_start = time.perf_counter()
+        if not hit:
+            self.encoded = None
+            self.encoded = self.encode(prefix, parts, suffix, key)
+        self.sync()
+        build_seconds = time.perf_counter() - build_start if not hit else 0.0
+        encoded = self.encoded
+        # History/question are intentionally never part of cached KV.
+        request = prefix[1:] + suffix if self.method == 'pcw' else suffix
+        if self.method.startswith('ape_'):
+            enable_attention_prefill_query('llama3.1-8b-instruct', self.model, float(param[0]), float(param[1]))
+            self.stage = 'query'
+        else:
+            self.stage = None
+        cache, logits = self.forward(request, cache=encoded.cache, position=encoded.next_position, logits=True)
+        eos = self.model.generation_config.eos_token_id
+        stops = set(eos if isinstance(eos, list) else [eos])
+        tokens = []
+        ttft = None
+        for step in range(row['max_new_tokens']):
+            token = int(logits.argmax(dim=-1).item())
+            self.sync()
+            if ttft is None:
+                ttft = time.perf_counter() - started
+            tokens.append(token)
+            if token in stops or step + 1 == row['max_new_tokens']:
+                break
+            cache, logits = self.forward([token], cache=cache, position=encoded.next_position + len(request) + step, logits=True)
+        self.sync()

+"""Definitions from LongBench, Lost in the Middle, and DialDoc21.
+
+Sources and downloaded originals are listed in SOURCES.md. All scoring joins by ID.
+"""
+import re
+import string
+from collections import Counter
+
+
+def normalize(text):
+    text = ''.join(c for c in text.lower() if c not in string.punctuation)
+    return ' '.join(re.sub(r'\b(a|an|the)\b', ' ', text).split())
+
+
+def qa_f1(prediction, answer):
+    pred, ref = normalize(prediction).split(), normalize(answer).split()
+    shared = sum((Counter(pred) & Counter(ref)).values())
+    if shared == 0:
+        return 0.0
+    precision, recall = shared / len(pred), shared / len(ref)
+    return 2 * precision * recall / (precision + recall)
+
+
+def best_subspan_em(prediction, answers):
+    return float(any(normalize(answer) in normalize(prediction) for answer in answers))
+
+
+def score(dataset, prediction, answers):
+    if not prediction.strip() or not answers:
+        raise ValueError('empty prediction or reference cannot be marked successful')
+    if dataset in {'qasper', 'hotpotqa', 'musique'}:
+        return max(qa_f1(prediction, a) for a in answers)
+    if dataset == 'litm':
+        return best_subspan_em(prediction, answers)
+    if dataset == 'doc2dial':
+        import sacrebleu
+        return sacrebleu.corpus_bleu([prediction], [[answers[0]]]).score
+    raise ValueError(f'unknown scoring protocol: {dataset}')
+
+
+def corpus_doc2dial(records, reference_by_id):
+    """Official sacrebleu defaults; explicit ID join avoids order-based mis-scoring."""
+    import sacrebleu
+    ids = [r['sample_id'] for r in records]
+    if len(set(ids)) != len(ids) or set(ids) != set(reference_by_id):
+        raise ValueError('Doc2Dial prediction/reference IDs do not align')
+    metric = sacrebleu.metrics.BLEU()
+    result = metric.corpus_score([r['answer'] for r in records], [[reference_by_id[i] for i in ids]])
+    return {'score': result.score, 'signature': str(metric.get_signature())}

+"""Predeclared 144 configurations and validation-only rule selection."""
+import math
+from itertools import product
+from statistics import mean
+
+LENGTHS = (4096, 8192, 16384)
+COUNTS = (4, 8, 16)
+PARAMS = tuple(product((0.5, 0.7, 0.9, 1.0), repeat=2))
+DEFAULT = (0.9, 0.9)
+DATASETS = ('qasper', 'hotpotqa', 'musique')
+
+
+def tie_key(param):
+    t, s = param
+    return ((t - 0.9) ** 2 + (s - 0.9) ** 2, t, s)
+
+
+def nearest(length, count):
+    if length <= 0 or count <= 0:
+        raise ValueError('L and N must be positive')
+    return min(product(LENGTHS, COUNTS), key=lambda cell: (
+        (math.log2(length) - math.log2(cell[0])) ** 2 + (math.log2(count) - math.log2(cell[1])) ** 2,
+        cell[0], cell[1]))
+
+
+def balanced_score(records, cells=None):
+    cells = list(cells or product(LENGTHS, COUNTS))
+    groups = {(d, l, n): [] for d in DATASETS for l, n in cells}
+    for r in records:
+        key = (r['dataset'], r['L'], r['N'])
+        if key in groups:
+            groups[key].append(r['score'])
+    missing = [key for key, values in groups.items() if not values]
+    if missing:
+        raise ValueError(f'incomplete balanced rule selection: {missing}')
+    return mean(mean(v) for v in groups.values())
+
+
+def select_exploration(records):
+    if any(r['split'] != 'explore' for r in records):
+        raise ValueError('only exploration records may choose candidates')
+    by_param = {p: [r for r in records if (r['T'], r['S']) == p] for p in PARAMS}
+    # Every parameter must use the same IDs in each cell.
+    identities = [sorted((r['dataset'], r['sample_id'], r['L'], r['N']) for r in rows) for rows in by_param.values()]
+    if not all(i == identities[0] for i in identities):
+        raise ValueError('unpaired exploration results')
+    global_best = min(PARAMS, key=lambda p: (-balanced_score(by_param[p]), tie_key(p)))
+    per_cell = {f'{l}/{n}': min(PARAMS, key=lambda p: (-balanced_score(by_param[p], [(l, n)]), tie_key(p))) for l, n in product(LENGTHS, COUNTS)}
+    return {'global': list(global_best), 'cells': {k: list(v) for k, v in per_cell.items()}}
+
+
+def freeze_rule(records, candidates):
+    if any(r['split'] != 'validation' for r in records):
+        raise ValueError('only validation may freeze the rule')
+    def rows(param, cell=None):
+        return [r for r in records if (r['T'], r['S']) == tuple(param) and (cell is None or (r['L'], r['N']) == cell)]
+    fixed_options = {DEFAULT, tuple(candidates['global'])}
+    # Check pairing separately per candidate / cell, including deduplicated runs.
+    for l, n in product(LENGTHS, COUNTS):
+        choices = fixed_options | {tuple(candidates['cells'][f'{l}/{n}'])}
+        ids = [sorted((r['dataset'], r['sample_id']) for r in rows(p, (l, n))) for p in choices]
+        if not ids[0] or not all(x == ids[0] for x in ids):
+            raise ValueError('unpaired validation results')
+    fixed = min(fixed_options, key=lambda p: (-balanced_score(rows(p)), tie_key(p)))
+    cells, scores = {}, {}
+    for l, n in product(LENGTHS, COUNTS):
+        p = tuple(candidates['cells'][f'{l}/{n}'])
+        best_score = balanced_score(rows(p, (l, n)), [(l, n)])
+        fixed_score = balanced_score(rows(fixed, (l, n)), [(l, n)])
+        cells[f'{l}/{n}'] = list(p if best_score > fixed_score else fixed)
+        scores[f'{l}/{n}'] = {'candidate': best_score, 'fixed': fixed_score}
+    return {'protocol': 'ape-v3-nine-cell', 'fixed': list(fixed), 'cells': cells, 'validation_scores': scores,
+            'mapping': 'Euclidean log2(L),log2(actual_N); ties smaller L then N; clamp boundaries'}
+
+
+def choose(rule, length, count):
+    l, n = nearest(length, count)
+    return tuple(rule['cells'][f'{l}/{n}'])

+def recovery_timing(result, seconds, peak=None):
+    result = dict(result, resume_cache_rebuild_seconds=seconds)
+    if seconds:
+        result['gpu_hot_ttft_after_rebuild_seconds'] = result['ttft_seconds']
+        result['gpu_hot_total_after_rebuild_seconds'] = result['total_seconds']
+        result['ttft_seconds'] += seconds
+        result['total_seconds'] += seconds
+        result['timing_condition'] = 'recovered_application_GPU_KV'
+        for name, value in (peak or {}).items():
+            result['resume_cache_rebuild_' + name] = value
+            result[name] = max(result.get(name) or 0, value)
+    return result

+def execute(bundle):

+    ledger = Ledger(bundle['ledger'])
